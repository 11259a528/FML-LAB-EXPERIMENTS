#4. Even and Odd Employee IDs Program

ids = [101, 102, 103, 104, 105, 106]

even = []
odd = []
for i in ids:
    if i % 2 == 0:
        even.append(i)
    else:
        odd.append(i)
print("Even IDs:", even)
print("Odd IDs:", odd)

#5. Count Positive and Negative Values Program
values = [20, -5, 15, -10, 30, -2]
positive = 0
negative = 0
for i in values:
    if i > 0:
        positive += 1
    elif i < 0:
        negative += 1
print("Positive Values:", positive)
print("Negative Values:", negative)

#6. Find Duplicate Customer IDs Program

customer_ids = [101, 102, 103, 102, 104, 105, 101]
duplicates = []
for i in customer_ids:
    if customer_ids.count(i) > 1 and i not in duplicates:
        duplicates.append(i)
print("Duplicate IDs:", duplicates)

#13. Age Group Classification Program

ages = [12, 25, 45, 67, 15]
for age in ages:
    if age < 18:
        print(age, "Child")
    elif age < 60:
        print(age, "Adult")
    else:
        print(age, "Senior Citizen")


#14. Electricity Bill Calculation Program

units = 250
if units <= 100:
    bill = units * 2
elif units <= 200:
    bill = 100 * 2 + (units - 100) * 3
else:
    bill = 100 * 2 + 100 * 3 + (units - 200) * 5
print("Electricity Bill:", bill)
